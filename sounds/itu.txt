--1.
CREATE TABLE Kategori (
    id_kategori INT AUTO_INCREMENT PRIMARY KEY,
    nama_kategori VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE Pelanggan (
    id_pelanggan INT AUTO_INCREMENT PRIMARY KEY,
    nama_pelanggan VARCHAR(100) NOT NULL,
    alamat VARCHAR(200),
    no_telp VARCHAR(15)
);

--2.
CREATE TABLE Pegawai (
    id_pegawai INT AUTO_INCREMENT PRIMARY KEY,
    nama_pegawai VARCHAR(100) NOT NULL,
    jabatan VARCHAR(30) NOT NULL,
    username VARCHAR(30) NOT NULL UNIQUE,
    password VARCHAR(100) NOT NULL
);

-- 3.
CREATE TABLE Produk (
    id_produk INT AUTO_INCREMENT PRIMARY KEY,
    nama_produk VARCHAR(100) NOT NULL,
    id_kategori INT,
    harga DECIMAL(10,2) NOT NULL DEFAULT 0 CHECK (harga >= 0),
    stok INT NOT NULL DEFAULT 0,
    satuan VARCHAR(20) DEFAULT 'pcs',
    FOREIGN KEY (id_kategori) REFERENCES Kategori(id_kategori)
);


-- 4. Buat tabel Transaksi (foreign key ke Pelanggan dan Pegawai)
CREATE TABLE Transaksi (
    id_transaksi INT AUTO_INCREMENT PRIMARY KEY,
    tanggal DATE NOT NULL,
    id_pelanggan INT,
    id_pegawai INT NOT NULL,
    FOREIGN KEY (id_pelanggan) REFERENCES Pelanggan(id_pelanggan),
    FOREIGN KEY (id_pegawai) REFERENCES Pegawai(id_pegawai)
);

-- 5. Buat tabel DetailTransaksi
CREATE TABLE DetailTransaksi (
    id_transaksi INT,
    id_produk INT,
    jumlah INT NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_transaksi) REFERENCES Transaksi(id_transaksi) ON DELETE CASCADE,
    FOREIGN KEY (id_produk) REFERENCES Produk(id_produk) ON DELETE RESTRICT
);

-- 6. Gunakan ALTER TABLE untuk menambah kolom email pada tabel Pelanggan
